using System.Collections;
using NUnit.Framework;
using UnityEngine;
using UnityEngine.TestTools;

/// <summary>
/// Play Mode tests for MOVE-03: Roll mechanic.
///
/// Tests:
/// 1. Roll grants invincibility (InvincibilityHandler.IsInvincible becomes true)
/// 2. Cooldown prevents a second roll until 0.8 real seconds have passed
/// 3. Cooldown drains in real time — not affected by timeScale
///
/// Manual test (not automated): Roll works visually in slow-motion at correct speed.
/// See 02-VALIDATION.md §Manual-Only Verifications for instructions.
/// </summary>
public class RollTests
{
    // -----------------------------------------------------------------------
    // MOVE-03: Roll grants i-frames
    // -----------------------------------------------------------------------

    [UnityTest]
    public IEnumerator Roll_GrantsInvincibilityOnActivation()
    {
        Time.timeScale      = 1f;
        Time.fixedDeltaTime = 0.02f;

        // Build player rig with all required components
        var playerGO = new GameObject("TestPlayer");
        playerGO.AddComponent<SpriteRenderer>();
        var rb = playerGO.AddComponent<Rigidbody2D>();
        rb.bodyType = RigidbodyType2D.Dynamic;
        var invincibility = playerGO.AddComponent<InvincibilityHandler>();
        playerGO.AddComponent<RollController>();

        yield return null; // let Awake complete

        Assert.IsFalse(invincibility.IsInvincible, "Should start non-invincible");

        // Trigger roll via reflection (simulates Shift key press without InputManager)
        var rollCtrl = playerGO.GetComponent<RollController>();
        var method = typeof(RollController)
            .GetMethod("RollCoroutine", System.Reflection.BindingFlags.NonPublic | System.Reflection.BindingFlags.Instance);
        rollCtrl.StartCoroutine((IEnumerator)method?.Invoke(rollCtrl, null));

        // One frame after roll starts — invincibility should be active
        yield return null;

        Assert.IsTrue(invincibility.IsInvincible,
            "InvincibilityHandler.IsInvincible should be true immediately after roll starts");

        // Wait for i-frame to expire (iFrameDuration=0.4s)
        yield return new WaitForSecondsRealtime(0.5f);

        Assert.IsFalse(invincibility.IsInvincible,
            "InvincibilityHandler.IsInvincible should be false after iFrameDuration expires");

        Object.DestroyImmediate(playerGO);
    }

    // -----------------------------------------------------------------------
    // MOVE-03: Roll cooldown prevents rapid re-activation
    // -----------------------------------------------------------------------

    [UnityTest]
    public IEnumerator Roll_CooldownPreventsImmediateReactivation()
    {
        Time.timeScale      = 1f;
        Time.fixedDeltaTime = 0.02f;

        var playerGO = new GameObject("TestPlayer");
        playerGO.AddComponent<SpriteRenderer>();
        var rb = playerGO.AddComponent<Rigidbody2D>();
        rb.bodyType = RigidbodyType2D.Dynamic;
        playerGO.AddComponent<InvincibilityHandler>();
        var rollCtrl = playerGO.AddComponent<RollController>();

        yield return null;

        // First roll
        var method = typeof(RollController)
            .GetMethod("RollCoroutine", System.Reflection.BindingFlags.NonPublic | System.Reflection.BindingFlags.Instance);
        rollCtrl.StartCoroutine((IEnumerator)method?.Invoke(rollCtrl, null));
        yield return null;

        // Check _isRolling or _cooldownRemaining — read via reflection
        var cooldownField = typeof(RollController)
            .GetField("_cooldownRemaining", System.Reflection.BindingFlags.NonPublic | System.Reflection.BindingFlags.Instance);
        float cooldownAfterRoll = (float)(cooldownField?.GetValue(rollCtrl) ?? 0f);

        Assert.Greater(cooldownAfterRoll, 0f,
            "_cooldownRemaining should be set to rollCooldown immediately after roll activates");

        Object.DestroyImmediate(playerGO);
    }

    // -----------------------------------------------------------------------
    // MOVE-03: Cooldown uses unscaled time (immune to timeScale)
    // -----------------------------------------------------------------------

    [UnityTest]
    public IEnumerator Roll_CooldownUsesUnscaledTime()
    {
        // This test verifies that the cooldown field depletes using unscaled time.
        // We set timeScale to 0.2x (slow-motion) and verify cooldown still decrements.

        Time.timeScale      = 1f;
        Time.fixedDeltaTime = 0.02f;

        var playerGO = new GameObject("TestPlayer");
        playerGO.AddComponent<SpriteRenderer>();
        var rb = playerGO.AddComponent<Rigidbody2D>();
        rb.bodyType = RigidbodyType2D.Dynamic;
        playerGO.AddComponent<InvincibilityHandler>();
        var rollCtrl = playerGO.AddComponent<RollController>();

        yield return null;

        // Trigger a roll to set the cooldown
        var rollMethod = typeof(RollController)
            .GetMethod("RollCoroutine", System.Reflection.BindingFlags.NonPublic | System.Reflection.BindingFlags.Instance);
        rollCtrl.StartCoroutine((IEnumerator)rollMethod?.Invoke(rollCtrl, null));
        yield return null;

        var cooldownField = typeof(RollController)
            .GetField("_cooldownRemaining", System.Reflection.BindingFlags.NonPublic | System.Reflection.BindingFlags.Instance);
        float cooldownAtStart = (float)(cooldownField?.GetValue(rollCtrl) ?? 0f);

        // Enter slow-motion (0.2x)
        Time.timeScale      = 0.2f;
        Time.fixedDeltaTime = 0.02f * Time.timeScale;

        // Wait 0.4 real seconds — at unscaled rate, cooldown should drop by ~0.4s
        yield return new WaitForSecondsRealtime(0.4f);

        float cooldownAfterWait = (float)(cooldownField?.GetValue(rollCtrl) ?? 0f);

        // If cooldown used Time.deltaTime (scaled), it would only drop by 0.4 * 0.2 = 0.08s
