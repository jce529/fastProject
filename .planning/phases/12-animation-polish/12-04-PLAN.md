[MenuItem("Fast/Phase12/Patch Player Animator (Whiff+Roll Triggers)")] public static void Run():
1. Load controller via AssetDatabase.LoadAssetAtPath<AnimatorController>(ControllerPath). If null, Debug.LogError and return.
2. var sm = controller.layers[0].stateMachine;
3. Whiff (D-05):
   - If !HasParameter(controller, "Whiff"), controller.AddParameter("Whiff", AnimatorControllerParameterType.Trigger).
   - var whiffState = FindState(sm, "Whiff"); if null:
     - Load AnimationClip at WhiffClipPath. If null, Debug.LogError and return (do not proceed with a null motion).
     - whiffState = sm.AddState("Whiff"); whiffState.motion = clip;
     - var anyToWhiff = sm.AddAnyStateTransition(whiffState); anyToWhiff.AddCondition(AnimatorConditionMode.If, 0, "Whiff"); anyToWhiff.duration = 0f; anyToWhiff.hasExitTime = false; anyToWhiff.canTransitionToSelf = false;
     - var idleState = FindState(sm, "Idle"); var whiffToIdle = whiffState.AddTransition(idleState); whiffToIdle.duration = 0f; whiffToIdle.hasExitTime = true; whiffToIdle.exitTime = 0.9f;
4. Roll (D-06):
   - If !HasParameter(controller, "Roll"), controller.AddParameter("Roll", AnimatorControllerParameterType.Trigger).
   - var rollState = FindState(sm, "Roll"); if rollState != null and !HasAnyStateTransitionTo(sm, rollState, "Roll"):
     - var anyToRoll = sm.AddAnyStateTransition(rollState); anyToRoll.AddCondition(AnimatorConditionMode.If, 0, "Roll"); anyToRoll.duration = 0f; anyToRoll.hasExitTime = false; anyToRoll.canTransitionToSelf = false;
   - Do NOT remove or modify the existing IsRolling bool-driven transitions into Roll (Idle/Walk/Sprint -> Roll) -- the new Roll trigger via AnyState coexists alongside them, it does not replace them.
5. EditorUtility.SetDirty(controller); AssetDatabase.SaveAssets(); Debug.Log("[PlayerAnimatorPatcher] Whiff/Roll 트리거+상태 패치 완료.");

Helper methods (private static):
- bool HasParameter(AnimatorController controller, string name) -- iterate controller.parameters, compare .name.
- AnimatorState FindState(AnimatorStateMachine sm, string name) -- iterate sm.states, compare child.state.name, return child.state.
- bool HasAnyStateTransitionTo(AnimatorStateMachine sm, AnimatorState target, string paramName) -- iterate sm.anyStateTransitions, check t.destinationState == target and any condition.parameter == paramName.
