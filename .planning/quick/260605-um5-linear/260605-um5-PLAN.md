```csharp
if (AttackTypeSelector.Selected == AttackType.Linear)
{
    // _linearAimDirection이 초기화 전(제로 벡터)이면 facing 방향으로 fallback
    Vector2 aimDir = _linearAimDirection.sqrMagnitude > 0.001f
        ? _linearAimDirection
        : (_spriteRenderer.flipX ? Vector2.left : Vector2.right);

    Vector2 toTarget = (targetPos - (Vector2)transform.position).normalized;
    float dot     = Vector2.Dot(aimDir, toTarget);
    float cosHalf = Mathf.Cos(fanHalfAngleDeg * Mathf.Deg2Rad);
    return dot >= cosHalf;
}
```
